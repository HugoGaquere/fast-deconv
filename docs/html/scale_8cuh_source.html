17__global__ void make_scales_kernel_half(float* sigmas, int scale_nrow, int scale_ncol_half,
18 int scale_ncol_full, int n_scales, float* scales)
19{
20 // col only ranges from 0 to ncol/2 (ncol/2+1 values)
21 // All frequencies are non-negative: no conditional branch needed
22
23 const int row = blockIdx.y * blockDim.y + threadIdx.y;
24 const int col = blockIdx.x * blockDim.x + threadIdx.x;
25 if (row >= scale_nrow || col >= scale_ncol_half) return;
26
27 const float freq_row = row < (scale_nrow + 1) / 2
28 ? static_cast<float>(row) / scale_nrow
29 : static_cast<float>(row - scale_nrow) / scale_nrow;
30 const float freq_col = static_cast<float>(col) / scale_ncol_full;
31
32 const float rhosq = freq_row * freq_row + freq_col * freq_col;
33
34 const int scale_size = scale_nrow * scale_ncol_half;
35 const int tid = row * scale_ncol_half + col;
36
37 for (int i = 0; i < n_scales; i++) {
38 const uint idx = tid + i * scale_size;
39 const float sigma = sigmas[i];
40 scales[idx] = exp(-2.0f * PI_SQUARRED * rhosq * sigma * sigma);
41 }
42}

44// Multiplies freq_dirty with each scale and normalizes by 1/N
45// freq_total = nrow * (ncol / 2 + 1) (half-complex from R2C)
46__global__ void multiply_batched_kernel(complex_type* freq_dirty, float* scales,
47 complex_type* scaled_dirty, int freq_total, int n_scales,
48 float norm)
49{
50 uint tid = blockIdx.x * blockDim.x + threadIdx.x;
51 if (tid >= freq_total) return;
52
53 const complex_type dirty_val = freq_dirty[tid];
54
55 for (int i = 0; i < n_scales; i++) {
56 const int idx = tid + i * freq_total;
57 const float scale_norm = scales[idx] * norm;
58 scaled_dirty[idx] = {dirty_val.x * scale_norm, dirty_val.y * scale_norm};
59 }
60}

62// Applies mask and optional abs to scaled_dirty in-place.
63// scaled_dirty: (n_scales, npix) modified in-place
64// mask: layout depends on mask_stride:
65// mask_stride=0: (npix,) shared mask across all scales
66// mask_stride=npix: (n_scales, npix) per-scale masks
67// True = masked (set to -inf)
68__global__ void apply_mask_kernel(float* scaled_dirty, const bool* mask, int npix, int n_scales,
69 int mask_stride, bool do_abs)
70{
71 const int tid = blockIdx.x * blockDim.x + threadIdx.x;
72 if (tid >= npix) return;
73
74 for (int s = 0; s < n_scales; s++) {
75 const int idx = s * npix + tid;
76 const bool masked = mask[s * mask_stride + tid];
77 if (masked) {
78 scaled_dirty[idx] = -INFINITY;
79 } else if (do_abs) {
80 scaled_dirty[idx] = fabsf(scaled_dirty[idx]);
81 }
82 }
83}

100// Finds the best scale and peak pixel via biased peak-finding.
101// scaled_dirty is modified in-place (mask + abs applied).
102// mask: device, True = masked. Either (npix,) shared or (n_scales, npix) per-scale.
103// bias: (n_scales,) host memory, multiplied with per-scale peaks to select best scale.
104// per_scale_mask: false = shared mask (npix,), true = per-scale masks (n_scales, npix)
105// Returns unbiased peak value and pixel coordinates.
106scale_selection_result scale_selection(const core::resources& resources,
107 const core::stream_resources& stream_res,
108 float* scaled_dirty, const bool* mask, const float* bias,
109 int n_scales, int nrow, int ncol, bool do_abs,
110 bool per_scale_mask)
111{
112 const auto cuda_stream = stream_res.cuda_stream;
113
114 const int npix = nrow * ncol;
115 const int mask_stride = per_scale_mask ? npix : 0;
116
117 // 1. Apply mask + abs in-place
118 apply_mask_kernel<<<CEIL_DIV(npix, 256), 256, 0, cuda_stream>>>(scaled_dirty, mask, npix,
119 n_scales, mask_stride, do_abs);
120
121 // 2. Build segment offsets [0, npix, 2*npix, ..., n_scales*npix]
122 int* d_offsets = resources.alloc_async<int>(n_scales + 1, stream_res);
123 std::vector<int> h_offsets(n_scales + 1);
124 for (int i = 0; i <= n_scales; i++) h_offsets[i] = i * npix;
125 CHECK_CUDA(cudaMemcpyAsync(d_offsets, h_offsets.data(), sizeof(int) * (n_scales + 1),
126 cudaMemcpyHostToDevice, cuda_stream));
127
128 // 3. CUB segmented argmax
129 using KVPair = cub::KeyValuePair<int, float>;
130 KVPair* d_peaks = resources.alloc_async<KVPair>(n_scales, stream_res);
131
132 size_t temp_bytes = 0;
133 CHECK_CUDA(cub::DeviceSegmentedReduce::ArgMax(nullptr, temp_bytes, scaled_dirty, d_peaks,
134 n_scales, d_offsets, d_offsets + 1, cuda_stream));
135
136 void* d_temp = resources.alloc_async(temp_bytes, stream_res);
137 CHECK_CUDA(cub::DeviceSegmentedReduce::ArgMax(d_temp, temp_bytes, scaled_dirty, d_peaks, n_scales,
138 d_offsets, d_offsets + 1, cuda_stream));
139
140 // 4. Copy per-scale peaks to host
141 std::vector<KVPair> h_peaks(n_scales);
142 CHECK_CUDA(cudaMemcpyAsync(h_peaks.data(), d_peaks, sizeof(KVPair) * n_scales,
143 cudaMemcpyDeviceToHost, cuda_stream));
144
145 stream_res.sync();
146
147 // Async cleanup
148 resources.free_async(d_offsets, stream_res);
149 resources.free_async(d_peaks, stream_res);
150 resources.free_async(d_temp, stream_res);
151
152 // 5. Biased scale selection on host
153 int best_scale = 0;
154 float best_biased = -INFINITY;
155 for (int s = 0; s < n_scales; s++) {
156 float biased = h_peaks[s].value * bias[s];
157 if (biased > best_biased) {
158 best_biased = biased;
159 best_scale = s;
160 }
161 }
162
163 int best_flat_idx = h_peaks[best_scale].key;
164 int best_row = best_flat_idx / ncol;
165 int best_col = best_flat_idx % ncol;
166
167 return {best_scale, best_row, best_col, h_peaks[best_scale].value};
168}

171{
172 const auto [npad_row, npad_col] = linalg::detail::compute_padding(nrow, ncol, padding);
173
174 scale_convole_ctx ctx;
175 ctx.img_nrow = nrow;
176 ctx.img_ncol = ncol;
177 ctx.padding_nrow = npad_row;
178 ctx.padding_ncol = npad_col;
179 ctx.img_padded_nrow = nrow + 2 * npad_row;
180 ctx.img_padded_ncol = ncol + 2 * npad_col;
181 ctx.freq_nrow = ctx.img_padded_nrow;
182 ctx.freq_ncol = ctx.img_padded_ncol / 2 + 1;

187 std::array<int, 2> fft_size{ctx.img_padded_nrow, ctx.img_padded_ncol};
188 CUFFT_CALL(cufftPlanMany(&ctx.plan_backward, 2, fft_size.data(), nullptr, 1, 0, nullptr, 1, 0,
189 CUFFT_C2R, n_scales));
190
191 return ctx;
192}

194// Convolves a dirty image with Gaussian scale kernels.
195// dirty: (nrow, ncol) real, device
196// sigmas: (n_scales,) Gaussian sigmas, device
197// out_scaled_dirty: (n_scales, nrow, ncol) real, device
198// Internally: pad+ifftshift → R2C → multiply with Gaussian scales → C2R → fftshift+crop
199void scale_convolve(const core::resources& resources, const core::stream_resources& stream_res,
200 const scale_convole_ctx& ctx, float* dirty, float* scales,
201 float* out_scaled_dirty, int n_scales)
202{
203 const int img_padded_total = ctx.img_padded_nrow * ctx.img_padded_ncol;
204 const int freq_total = ctx.freq_nrow * ctx.freq_ncol;
205 const int freq_scales_total = freq_total * n_scales;
206
207 cudaStream_t cuda_stream = stream_res.cuda_stream;
208 CUFFT_CALL(cufftSetStream(ctx.plan_forward, cuda_stream));
209 CUFFT_CALL(cufftSetStream(ctx.plan_backward, cuda_stream));
210
211 // Allocate temporaries
212 float* dirty_padded = resources.alloc_async<float>(img_padded_total, stream_res);
213 complex_type* dirty_freq = resources.alloc_async<complex_type>(freq_total, stream_res);
214 complex_type* scaled_dirty_freq =
215 resources.alloc_async<complex_type>(freq_scales_total, stream_res);
216 float* scaled_dirty = resources.alloc_async<float>(img_padded_total * n_scales, stream_res);
217
218 stream_res.sync();
219
220 // Pad + ifftshift dirty image
221 linalg::detail::pad_ifftshift(dirty, dirty_padded, ctx.img_nrow, ctx.img_ncol,
222 ctx.img_padded_nrow, ctx.img_padded_ncol, ctx.padding_nrow,
223 ctx.padding_ncol, cuda_stream);
224
225 // Forward R2C: real (nrow, ncol) -> half-complex (nrow, ncol/2+1)
226 CUFFT_CALL(cufftExecR2C(ctx.plan_forward, dirty_padded, dirty_freq));
227
228 // Element-wise multiply with scales + 1/N normalization
229 float norm = 1.0f / static_cast<float>(img_padded_total);
230 multiply_batched_kernel<<<CEIL_DIV(freq_total, 256), 256, 0, cuda_stream>>>(
231 dirty_freq, scales, scaled_dirty_freq, freq_total, n_scales, norm);
232
233 // Inverse C2R: half-complex -> real per batch
234 CUFFT_CALL(cufftExecC2R(ctx.plan_backward, scaled_dirty_freq, scaled_dirty));
235
236 // Fftshift + crop back to original size
237 linalg::detail::fftshift_crop(scaled_dirty, out_scaled_dirty, ctx.img_nrow, ctx.img_ncol,
238 ctx.img_padded_nrow, ctx.img_padded_ncol, ctx.padding_nrow,
239 ctx.padding_ncol, n_scales, cuda_stream);
240
241 // Cleanup
242 resources.free_async(dirty_padded, stream_res);
243 resources.free_async(dirty_freq, stream_res);
244 resources.free_async(scaled_dirty_freq, stream_res);
245 resources.free_async(scaled_dirty, stream_res);
246
247 stream_res.sync();
248}
