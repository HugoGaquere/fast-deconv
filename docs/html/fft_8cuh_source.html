23// Pads and ifftshifts a 2D image in one pass.
24// Input: (nx, ny) real, origin at center
25// Output: (px, py) real, origin at (0,0), zero-padded
26// Output must be zeroed before launch.
27__global__ void pad_ifftshift_kernel(const float* input, float* output, int nx, int ny, int px,
28 int py, int npad_x, int npad_y)
29{
30 const int row = blockIdx.y * blockDim.y + threadIdx.y;
31 const int col = blockIdx.x * blockDim.x + threadIdx.x;
32 if (row >= nx || col >= ny) return;
33
34 // Position in padded array (input centered)
35 const int pad_row = row + npad_x;
36 const int pad_col = col + npad_y;
37
38 // ifftshift: shift by ceil(N/2) = (N+1)/2
39 const int out_row = (pad_row + (px + 1) / 2) % px;
40 const int out_col = (pad_col + (py + 1) / 2) % py;
41
42 output[out_row * py + out_col] = input[row * ny + col];
43}

45// Fftshifts and crops a batched 2D image in one pass.
46// Input: (n_batch, px, py) real, origin at (0,0) (FFT output)
47// Output: (n_batch, nx, ny) real, origin at center, cropped
48__global__ void fftshift_crop_kernel(const float* input, float* output, int nx, int ny, int px,
49 int py, int npad_x, int npad_y, int n_batch)
50{
51 const int row = blockIdx.y * blockDim.y + threadIdx.y;
52 const int col = blockIdx.x * blockDim.x + threadIdx.x;
53 if (row >= nx || col >= ny) return;
54
55 const int src_row = (row + npad_x + (px + 1) / 2) % px;
56 const int src_col = (col + npad_y + (py + 1) / 2) % py;
57
58 const int out_stride = nx * ny;
59 const int in_stride = px * py;
60 const int out_idx = row * ny + col;
61 const int in_idx = src_row * py + src_col;
62
63 for (int b = 0; b < n_batch; b++) {
64 output[b * out_stride + out_idx] = input[b * in_stride + in_idx];
65 }
66}
