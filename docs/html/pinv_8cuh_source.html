17// Pseudo-inverse via normal equations: A+ = (A^T A)^{-1} A^T
18// Uses Cholesky factorization for the N×N system (A^T A is SPD for full column rank A).
19// A is M×N (row-major), A_pinv is N×M (row-major). Requires M >= N.
20// Single-block kernel; all computation in shared memory.
21//
22// Dynamic shared memory layout (floats):
23// As : M * N — copy of A
24// AtA : N * N — A^T A
25// At : N * M — A^T
26// L : N * N — Cholesky factor (lower triangular)
27static __global__ void pinv_kernel(const float* __restrict__ A,
28 float* __restrict__ A_pinv,
29 int M,
30 int N)
31{
32 extern __shared__ float smem[];
33
34 float* As = smem;
35 float* AtA = As + M * N;
36 float* At = AtA + N * N;
37 float* L = At + N * M;
38
39 const int tid = threadIdx.x;
40
41 // ---------------------------------------------------------------
42 // 1. Load A into shared memory
43 // ---------------------------------------------------------------
44 for (int i = tid; i < M * N; i += blockDim.x)
45 As[i] = A[i];
46
47 __syncthreads();
48
49 // ---------------------------------------------------------------
50 // 2. AtA = A^T * A (N×N, symmetric)
51 // AtA(i,j) = sum_k A(k,i) * A(k,j)
52 // ---------------------------------------------------------------
53 for (int idx = tid; idx < N * N; idx += blockDim.x) {
54 const int i = idx / N;
55 const int j = idx % N;
56 float sum = 0.0f;
57 for (int k = 0; k < M; ++k)
58 sum += As[k * N + i] * As[k * N + j];
59 AtA[i * N + j] = sum;
60 }
61
62 __syncthreads();
63
64 // ---------------------------------------------------------------
65 // 3. At = A^T (N×M)
66 // At(i,j) = A(j,i)
67 // ---------------------------------------------------------------
68 for (int idx = tid; idx < N * M; idx += blockDim.x) {
69 const int i = idx / M;
70 const int j = idx % M;
71 At[i * M + j] = As[j * N + i];
72 }
73
74 __syncthreads();
75
76 // ---------------------------------------------------------------
77 // 4. Cholesky factorization: AtA = L * L^T (serial, N is small)
78 // ---------------------------------------------------------------
79 if (tid == 0) {
80 for (int i = 0; i < N; ++i) {
81 // Zero upper triangle of L row
82 for (int j = i + 1; j < N; ++j)
83 L[i * N + j] = 0.0f;
84
85 for (int j = 0; j <= i; ++j) {
86 float sum = AtA[i * N + j];
87 for (int k = 0; k < j; ++k)
88 sum -= L[i * N + k] * L[j * N + k];
89 if (i == j)
90 L[i * N + j] = sqrtf(sum);
91 else
92 L[i * N + j] = sum / L[j * N + j];
93 }
94 }
95 }
96
97 __syncthreads();
98
99 // ---------------------------------------------------------------
100 // 5. Solve L L^T X = A^T for X = A+ (N×M)
101 // Each thread handles one or more columns of the RHS.
102 // ---------------------------------------------------------------
103 for (int col = tid; col < M; col += blockDim.x) {
104 // Forward substitution: L y = At[:, col]
105 float y[PINV_MAX_COLS];
106 for (int i = 0; i < N; ++i) {
107 float sum = At[i * M + col];
108 for (int k = 0; k < i; ++k)
109 sum -= L[i * N + k] * y[k];
110 y[i] = sum / L[i * N + i];
111 }
112
113 // Back substitution: L^T x = y
114 float x[PINV_MAX_COLS];
115 for (int i = N - 1; i >= 0; --i) {
116 float sum = y[i];
117 for (int k = i + 1; k < N; ++k)
118 sum -= L[k * N + i] * x[k];
119 x[i] = sum / L[i * N + i];
120 }
121
122 // Write result to A_pinv[:, col]
123 for (int i = 0; i < N; ++i)
124 A_pinv[i * M + col] = x[i];
125 }
126}

128} // namespace detail

130// Host launcher for the pseudo-inverse kernel.
131// Computes A+ = (A^T A)^{-1} A^T via Cholesky factorization.
132// A is (rows x cols) row-major, A_pinv is (cols x rows) row-major.
133// Requires rows >= cols and A to have full column rank.
134inline void pinv(const core::device_span2d<float>& A,
135 core::device_span2d<float>& A_pinv,
136 int rows,
137 int cols,
138 core::stream_resources& resources)
139{
140 assert(rows >= cols && "pinv requires rows >= cols (overdetermined system)");
141 assert(cols <= PINV_MAX_COLS && "cols exceeds PINV_MAX_COLS for per-thread solve arrays");
142
143 const int block_size = (rows > 256) ? 256 : ((rows + 31) / 32) * 32;
144
145 // Dynamic shared memory: As(M*N) + AtA(N*N) + At(N*M) + L(N*N)
146 const size_t smem_bytes = static_cast<size_t>(
147 rows * cols + cols * cols + cols * rows + cols * cols) * sizeof(float);
148
149 detail::pinv_kernel<<<1, block_size, smem_bytes, resources.cuda_stream>>>(
150 A.data_handle(), A_pinv.data_handle(), rows, cols);
151 CHECK_LAST_CUDA_ERROR();
152}
