20struct overlap_region {
21 int ax0, ay0, bx0, by0; // Top-left in A and B
22 int w, h; // Overlap size
23 int lda, ldb; // A and B row stride in elements
24};

41__host__ __device__ inline overlap_region compute_overlap_region(int y_center, int x_center,
42 int a_height, int a_width,
43 int b_height, int b_width)
44{
45 // X axis
46 const int b_x0 = x_center - b_width / 2;
47 const int a_x0 = b_x0 > 0 ? b_x0 : 0;
48 const int b_x_offset = a_x0 - b_x0;
49
50 const int b_x1 = x_center + b_width / 2;
51 const int a_x1_inclusive = b_x1 < a_width - 1 ? b_x1 : a_width - 1;
52 const int b_x1_offset = b_x1 - a_x1_inclusive;
53 const int b_x1_aligned = b_width - b_x1_offset;
54
55 // Y axis
56 const int b_y0 = y_center - b_height / 2;
57 const int a_y0 = b_y0 > 0 ? b_y0 : 0;
58 const int b_y_offset = a_y0 - b_y0;
59
60 const int b_y1 = y_center + b_height / 2;
61 const int a_y1_inclusive = b_y1 < a_height - 1 ? b_y1 : a_height - 1;
62 const int b_y1_offset = b_y1 - a_y1_inclusive;
63 const int b_y1_aligned = b_height - b_y1_offset;
64
65 // Compute overlap dimensions
66 const int w = b_x1_aligned - b_x_offset;
67 const int h = b_y1_aligned - b_y_offset;
68
69 return overlap_region{a_x0, a_y0, b_x_offset, b_y_offset, w, h, a_width, b_width};
70}

101template <int BLOCK_SIZE>
102__device__ void grid_argmax(
103 cg::grid_group grid, IndexedValue thread_data, IndexedValue* __restrict__ block_scratch,
104 typename cub::BlockReduce<IndexedValue, BLOCK_SIZE>::TempStorage& temp_storage)
105{
106 using BlockReduce = cub::BlockReduce<IndexedValue, BLOCK_SIZE>;
107
108 auto block = cg::this_thread_block();
109 const int tid = block.thread_rank();
110
111 // Phase 1: per-block reduction
112 IndexedValue block_max = BlockReduce(temp_storage).Reduce(thread_data, ArgMaxOp());
113
114 if (tid == 0) {
115 block_scratch[blockIdx.x] = block_max;
116 }
117
118 grid.sync();
119
120 // Phase 2: block 0 reduces across blocks
121 if (blockIdx.x == 0) {
122 IndexedValue local{-FLT_MAX, -1};
123 const auto num_blocks = grid.num_blocks();
124 for (unsigned i = tid; i < num_blocks; i += block.num_threads()) {
125 local = ArgMaxOp{}(local, block_scratch[i]);
126 }
127
128 IndexedValue final_max = BlockReduce(temp_storage).Reduce(local, ArgMaxOp());
129
130 if (tid == 0) {
131 block_scratch[0] = final_max;
132 }
133 }
134
135 grid.sync();
136}

138// residual: device float nrow x ncol
139// psfs: device float n_facet x psf_nrow x psf_ncol
140// map_pixels_facets: device int nrow x ncol
141// gains: device float n_facet
142template <int BLOCK_SIZE>
143__global__ void clean_minor_cycles_kernel(float* residual, float* psfs, int* map_pixels_facets,
144 float* gains, int nrow, int ncol, int psf_nrow,
145 int psf_ncol, int n_facet,
146 IndexedValue* __restrict__ block_scratch, float threshold,
147 int max_iter)
148{
149 const int tid = blockIdx.x * blockDim.x + threadIdx.x;
150 const int stride = blockDim.x * gridDim.x;
151 const int n = nrow * ncol;
152
153 auto grid = cg::this_grid();
154
155 using BlockReduce = cub::BlockReduce<IndexedValue, BLOCK_SIZE>;
156 __shared__ typename BlockReduce::TempStorage temp_storage;
157
158 // Grid-stride loop: each thread reduces its chunk locally first
159 IndexedValue my_data{-FLT_MAX, -1};
160 for (int i = tid; i < n; i += stride) {
161 if (residual[i] > my_data.value) {
162 my_data = {residual[i], i};
163 }
164 }
165 grid_argmax<BLOCK_SIZE>(grid, my_data, block_scratch, temp_storage);
166
167 IndexedValue peak = block_scratch[0];
168
169 int n_iter = 0;
170 while (peak.value > threshold && n_iter < max_iter) {
171 // Phase 1: Unravel peak index + compute ROI + gain factor
172 // All threads compute identically (broadcast reads from L2)
173 auto [peak_row, peak_col] = unravel_index_2D(peak.index, ncol);
174 int facet_id = map_pixels_facets[peak_row * ncol + peak_col];
175 overlap_region ovr = compute_overlap_region(peak_row, peak_col, nrow, ncol, psf_nrow, psf_ncol);
176 float factor = gains[facet_id] * peak.value;
177 const float* psf_facet = psfs + facet_id * psf_nrow * psf_ncol;
178
179 // Phase 2: Grid-stride PSF subtraction
180 int roi_size = ovr.w * ovr.h;
181 for (int i = tid; i < roi_size; i += stride) {
182 int local_y = i / ovr.w;
183 int local_x = i % ovr.w;
184 int a_offset = (ovr.ay0 + local_y) * ovr.lda + (ovr.ax0 + local_x);
185 int b_offset = (ovr.by0 + local_y) * ovr.ldb + (ovr.bx0 + local_x);
186 float r = residual[a_offset];
187 if (r > -FLT_MAX) // skip masked pixels
188 residual[a_offset] = r - psf_facet[b_offset] * factor;
189 }
190
191 // Phase 3: Sync + full re-scan argmax
192 grid.sync();
193
194 IndexedValue my_data_iter{-FLT_MAX, -1};
195 for (int i = tid; i < n; i += stride) {
196 if (residual[i] > my_data_iter.value) {
197 my_data_iter = {residual[i], i};
198 }
199 }
200 grid_argmax<BLOCK_SIZE>(grid, my_data_iter, block_scratch, temp_storage);
201
202 peak = block_scratch[0];
203 n_iter++;
204 }
205}

209 core::device_span4d<float>& psfs_2,

211 core::device_span2d<float>& gains, int scale_idx, float threshold,
212 int max_iter)

214 int nrow = residual.extent(0);
215 int ncol = residual.extent(1);

269__global__ void psf_subtract_kernel(float* __restrict__ residual, const float* __restrict__ psf,
270 overlap_region ovr, float factor)
271{
272 const int n = ovr.w * ovr.h;
273 for (int idx = blockIdx.x * blockDim.x + threadIdx.x; idx < n; idx += blockDim.x * gridDim.x) {
274 const int local_y = idx / ovr.w;
275 const int local_x = idx % ovr.w;
276
277 const int a_offset = (ovr.ay0 + local_y) * ovr.lda + (ovr.ax0 + local_x);
278 const int b_offset = (ovr.by0 + local_y) * ovr.ldb + (ovr.bx0 + local_x);
279
280 residual[a_offset] -= psf[b_offset] * factor;
281 }
282}

284// Compute A[f, o] = xdes[f, o] * sqrt(jones_norm[f, peak_row, peak_col]) * sqrt(weights[f])
285// jones_norm row-major: [n_freq, nrow, ncol]
286// xdes row-major: [n_freq, n_order]
287// weights row-major: [n_freq]
288// A row-major: [n_freq, n_order]
289__global__ void compute_spectral_matrix_kernel(float* __restrict__ A,
290 const float* __restrict__ xdes,
291 const float* __restrict__ jones_norm,
292 const float* __restrict__ weights, int n_freq,
293 int n_order, int nrow, int ncol, int peak_row,
294 int peak_col)
295{
296 const int n = n_freq * n_order;
297 const int spatial_stride = nrow * ncol;
298 const int peak_offset = peak_row * ncol + peak_col;
299 for (int idx = blockIdx.x * blockDim.x + threadIdx.x; idx < n; idx += blockDim.x * gridDim.x) {
300 const int f = idx / n_order;
301 float jn = jones_norm[f * spatial_stride + peak_offset];
302 A[idx] = xdes[idx] * sqrtf(jn) * sqrtf(weights[f]);
303 }
304}
305
306// Compute A_pinv = inv(A^T A) @ A^T
307// A: row-major [n_freq, n_order] (device)
308// A_pinv: col-major [n_order, n_freq] output (device), pre-allocated
309//
310// cuBLAS sees row-major A as col-major A_cm = A^T [n_order, n_freq]
311// Step 1: G = A_cm @ A_cm^T = A^T A [n_order, n_order]
312// Step 2: G_inv = inv(G) [n_order, n_order] (matinvBatched)
313// Step 3: A_pinv = G_inv @ A_cm = inv(A^T A) @ A^T [n_order, n_freq]
314void compute_pseudo_inverse(const core::resources& resources,
315 const core::stream_resources& stream_res, const float* d_A,
316 float* d_Apinv, int n_freq, int n_order)
317{
318 auto handle = stream_res.cublas_handle;
319 const float alpha = 1.0f;
320 const float beta = 0.0f;
321
322 float* d_G = resources.alloc_async<float>(n_order * n_order, stream_res);
323 float* d_Ginv = resources.alloc_async<float>(n_order * n_order, stream_res);
324 int* d_info = resources.alloc_async<int>(1, stream_res);
325 float** d_G_ptrs = resources.alloc_async<float*>(1, stream_res);
326 float** d_Ginv_ptrs = resources.alloc_async<float*>(1, stream_res);
327
328 // Step 1: G = A^T @ A
329 CHECK_CUBLAS(cublasSgemm(handle, CUBLAS_OP_N, CUBLAS_OP_T, n_order, n_order, n_freq, &alpha, d_A,
330 n_order, d_A, n_order, &beta, d_G, n_order));
331
332 // Step 2: G_inv = inv(G)
333 // matinvBatched expects device arrays of device pointers;
334 // copy host-side pointer values to device so cuBLAS can read them
335 CHECK_CUDA(cudaMemcpyAsync(d_G_ptrs, &d_G, sizeof(float*), cudaMemcpyHostToDevice,
336 stream_res.cuda_stream));
337 CHECK_CUDA(cudaMemcpyAsync(d_Ginv_ptrs, &d_Ginv, sizeof(float*), cudaMemcpyHostToDevice,
338 stream_res.cuda_stream));
339 CHECK_CUBLAS(
340 cublasSmatinvBatched(handle, n_order, d_G_ptrs, n_order, d_Ginv_ptrs, n_order, d_info, 1));
341
342 // Step 3: A_pinv = G_inv @ A^T
343 CHECK_CUBLAS(cublasSgemm(handle, CUBLAS_OP_N, CUBLAS_OP_N, n_order, n_freq, n_order, &alpha,
344 d_Ginv, n_order, d_A, n_order, &beta, d_Apinv, n_order));
345
346 resources.free_async(d_Ginv_ptrs, stream_res);
347 resources.free_async(d_G_ptrs, stream_res);
348 resources.free_async(d_info, stream_res);
349 resources.free_async(d_Ginv, stream_res);
350 resources.free_async(d_G, stream_res);
351}

353// Fused kernel: compute spectral coefficients at peak location
354// Step 1: wy[f] = sqrt_w[f] * dirty[f, peak_row, peak_col]
355// Step 2: compact[o] = sum_f A_pinv(o, f) * wy[f] (one thread per order, loop over freq)
356// Step 3: per_chan[f] = sum_o A(f, o) * compact[o] (one thread per freq, loop over order)
357//
358// A_pinv: col-major [n_order, n_freq] (from compute_pseudo_inverse)
359// A: row-major [n_freq, n_order] (from compute_spectral_matrix_kernel)
360constexpr int SPECTRAL_BLOCK_SIZE = 128;
361__global__ void compute_spectral_coeffs_kernel(float* __restrict__ compact_out,
362 float* __restrict__ per_chan_out,
363 const float* __restrict__ dirty,
364 const float* __restrict__ weights,
365 const float* __restrict__ A_pinv,
366 const float* __restrict__ A, int n_freq, int n_order,
367 int nrow, int ncol, int peak_row, int peak_col)
368{
369 extern __shared__ float smem[];
370 float* s_wy = smem; // [n_freq]
371 float* s_compact = s_wy + n_freq; // [n_order]
372
373 const int tid = threadIdx.x;
374 const int spatial_stride = nrow * ncol;
375 const int peak_offset = peak_row * ncol + peak_col;
376
377 // Step 1: build weighted vector at peak pixel
378 for (int f = tid; f < n_freq; f += SPECTRAL_BLOCK_SIZE) {
379 s_wy[f] = sqrtf(weights[f]) * dirty[f * spatial_stride + peak_offset];
380 }
381 __syncthreads();
382
383 // Step 2: compact[o] = A_pinv[o, :] . wy
384 // A_pinv col-major [n_order, n_freq]: element (o, f) at [f * n_order + o]
385 // One thread per order, each loops over all frequencies
386 if (tid < n_order) {
387 float sum = 0.0f;
388 for (int f = 0; f < n_freq; f++) {
389 sum += A_pinv[f * n_order + tid] * s_wy[f];
390 }
391 s_compact[tid] = sum;
392 compact_out[tid] = sum;
393 }
394 __syncthreads();
395
396 // Step 3: per_chan[f] = A[f, :] . compact
397 // A row-major [n_freq, n_order]: element (f, o) at [f * n_order + o]
398 // One thread per freq, each loops over all orders
399 for (int f = tid; f < n_freq; f += SPECTRAL_BLOCK_SIZE) {
400 float sum = 0.0f;
401 for (int o = 0; o < n_order; o++) {
402 sum += A[f * n_order + o] * s_compact[o];
403 }
404 per_chan_out[f] = sum;
405 }
406}

408void compute_spectral_coeffs(const core::resources& resources,
409 const core::stream_resources& stream_res, float* compact_out,
410 float* per_chan_out, const float* dirty, const float* weights,
411 const float* A_pinv, const float* A, int n_freq, int n_order, int nrow,
412 int ncol, int peak_row, int peak_col)
413{

417 compact_out, per_chan_out, dirty, weights, A_pinv, A, n_freq, n_order, nrow, ncol, peak_row,

421// Subtract PSF from dirty across all frequencies at peak location
422// dirty[f, y, x] -= per_chan[f] * gain * psf[f, y, x] (within overlap region)
423__global__ void spectral_psf_subtract_kernel(float* __restrict__ dirty,
424 const float* __restrict__ psf,
425 const float* __restrict__ per_chan, overlap_region ovr,
426 float gain, int n_freq, int dirty_spatial_stride,
427 int psf_spatial_stride)
428{
429 const int roi_size = ovr.w * ovr.h;
430 const int total = n_freq * roi_size;
431
432 for (int idx = blockIdx.x * blockDim.x + threadIdx.x; idx < total;
433 idx += blockDim.x * gridDim.x) {
434 const int f = idx / roi_size;
435 const int spatial = idx % roi_size;
436 const int ly = spatial / ovr.w;
437 const int lx = spatial % ovr.w;
438
439 const int a_offset = f * dirty_spatial_stride + (ovr.ay0 + ly) * ovr.lda + (ovr.ax0 + lx);
440 const int b_offset = f * psf_spatial_stride + (ovr.by0 + ly) * ovr.ldb + (ovr.bx0 + lx);
441
442 dirty[a_offset] -= per_chan[f] * gain * psf[b_offset];
443 }
444}

446// Full spectral component subtraction pipeline:
447// 1. A = sqrt(jones_norm[f, peak]) * WXdes[f, o]
448// 2. A_pinv = inv(A^T A) @ A^T
449// 3. wy = sqrt_w * dirty[f, peak] → compact = A_pinv @ wy → per_chan = A @ compact
450// 4. dirty[f] -= per_chan[f] * gain * psf[f] (within overlap region)
451void subtract_component(const core::resources& resources, const core::stream_resources& stream_res,
452 float* dirty, const float* psf, const float* xdes, const float* jones_norm,
453 const float* weights, float gain, int n_freq, int n_order, int nrow,
454 int ncol, int psf_nrow, int psf_ncol, int peak_row, int peak_col)
455{
456 auto cuda_stream = stream_res.cuda_stream;
457
458 // Allocate temporaries
459 float* d_A = resources.alloc_async<float>(n_freq * n_order, stream_res);
460 float* d_A_pinv = resources.alloc_async<float>(n_order * n_freq, stream_res);
461 float* d_compact = resources.alloc_async<float>(n_order, stream_res);
462 float* d_per_chan = resources.alloc_async<float>(n_freq, stream_res);
463
464 // Step 1: build spectral matrix A
465 const int n_A = n_freq * n_order;
466 compute_spectral_matrix_kernel<<<1, n_A, 0, cuda_stream>>>(
467 d_A, xdes, jones_norm, weights, n_freq, n_order, nrow, ncol, peak_row, peak_col);
468
469 // Step 2: pseudo-inverse
470 compute_pseudo_inverse(resources, stream_res, d_A, d_A_pinv, n_freq, n_order);
471
472 // Step 3: spectral coefficients
473 compute_spectral_coeffs(resources, stream_res, d_compact, d_per_chan, dirty, weights, d_A_pinv,
474 d_A, n_freq, n_order, nrow, ncol, peak_row, peak_col);
475
476 // Step 4: subtract PSF from dirty
477 overlap_region ovr = compute_overlap_region(peak_row, peak_col, nrow, ncol, psf_nrow, psf_ncol);
478 const int total = n_freq * ovr.w * ovr.h;
479 spectral_psf_subtract_kernel<<<CEIL_DIV(total, 256), 256, 0, cuda_stream>>>(
480 dirty, psf, d_per_chan, ovr, gain, n_freq, nrow * ncol, psf_nrow * psf_ncol);
481
482 // Free temporaries
483 resources.free_async(d_per_chan, stream_res);
484 resources.free_async(d_compact, stream_res);
485 resources.free_async(d_A_pinv, stream_res);
486 resources.free_async(d_A, stream_res);
487}

535
536 FD_LOG_INFO("minor_loop: initial_peak={:.8f} threshold={:.8f} max_iter={}", h_peak.value,
537 threshold, params.max_subminor_iter);
538 FD_LOG_DEBUG("minor_loop: grid=[{},{}] psf=[{},{}] n_freq={} n_order={}", nrow, ncol, psf_nrow,
539 psf_ncol, n_freq, n_order);
540
541 int n_iter = 0;
542 while (h_peak.value > threshold && n_iter < params.max_subminor_iter) {
543 auto [peak_row, peak_col] = unravel_index_2D(h_peak.key, ncol);
544 int facet_idx = ctx.map_pixel_facet(peak_row, peak_col);
545 float gain = ctx.gains(scale_idx, facet_idx);
546 float factor = gain * h_peak.value;
547
548 FD_LOG_DEBUG("minor_loop: iter={} peak={:.8f} at ({},{}) facet={} gain={:.4f}", n_iter,
549 h_peak.value, peak_row, peak_col, facet_idx, gain);
550
551 // Stream 1: PSF subtraction from mean dirty
552 const float* psf_2_ptr = psfs_2.data_handle() + psfs_2.mapping()(scale_idx, facet_idx, 0, 0);
553 overlap_region ovr = compute_overlap_region(peak_row, peak_col, nrow, ncol, psf_nrow, psf_ncol);
554 psf_subtract_kernel<<<CEIL_DIV(ovr.w * ovr.h, 256), 256, 0, cuda_stream>>>(
555 mean_residual_ptr, psf_2_ptr, ovr, factor);
556
557 // Stream 2: subtract component from dirty
558 const float* psf_ptr = psfs.data_handle() + psfs.mapping()(scale_idx, facet_idx, 0, 0, 0, 0);
559 subtract_component(resources, stream_res_2, residual_ptr, psf_ptr, xdes_ptr, jones_norm_ptr,
560 weights_ptr, gain, n_freq, n_order, nrow, ncol, psf_nrow, psf_ncol, peak_row,
561 peak_col);
562
563 // Stream 1: Find peak
564 cub::DeviceReduce::ArgMax(d_temp, temp_storage_bytes, mean_residual_ptr, d_argmax_out, n,
565 cuda_stream);
566 CHECK_CUDA(cudaMemcpyAsync(&h_peak, d_argmax_out, sizeof(KVPair), cudaMemcpyDeviceToHost,
567 cuda_stream));
568
569 stream_res.sync();
570 stream_res_2.sync();
571
572 n_iter++;
573 }
574
575 FD_LOG_INFO("minor_loop: finished after {} iterations, final_peak={:.8f}", n_iter, h_peak.value);
576
577 // Cleanup
578 resources.free_async(d_temp, stream_res);
579 resources.free_async(d_argmax_out, stream_res);
580 stream_res.sync();
581}
